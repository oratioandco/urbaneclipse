 * `bounds` additionally allows a sweep to reject instants where the body is nowhere
 * near the landmark without projecting anything at all.
 */
export interface SilhouetteDirections {
  outline: AzAlt[];
  bounds: { azMin: number; azMax: number; altMin: number; altMax: number };
}

export function buildSilhouetteDirections(
  observer: ObserverGeodetic,
  landmark: RevolutionLandmark,
  steps = 96,
): SilhouetteDirections {
  const total = landmarkHeight(landmark);
  const empty = { outline: [], bounds: { azMin: 0, azMax: 0, altMin: 0, altMax: 0 } };
  if (total <= 0) return empty;

  const baseEllipsoidal = orthometricToEllipsoidal(landmark.baseOrthometric);

  const left: AzAlt[] = [];
  const right: AzAlt[] = [];
  let azMin = Infinity;
  let azMax = -Infinity;
  let altMin = Infinity;
  let altMax = -Infinity;

  for (let i = 0; i <= steps; i++) {
    const hAgl = (total * i) / steps;
    const axis = azAltTo(observer, landmark.lat, landmark.lon, baseEllipsoidal + hAgl);
    const r = radiusAtHeight(landmark, hAgl);

    // Horizontal angular half-width of the limb at this height.
    const ratio = Math.min(1, r / Math.max(axis.rangeM, 1e-9));
    const halfWidthDeg = Math.asin(ratio) * DEG;

    // Offset in azimuth. Divide by cos(alt) so that after the tangent-plane
    // projection's cos(alt) compression the on-sky width is correct.
    const cosAlt = Math.max(Math.cos(axis.alt * RAD), 1e-9);
    const dAz = halfWidthDeg / cosAlt;

    left.push({ az: axis.az - dAz, alt: axis.alt });
    right.push({ az: axis.az + dAz, alt: axis.alt });

    azMin = Math.min(azMin, axis.az - dAz);
    azMax = Math.max(azMax, axis.az + dAz);
    altMin = Math.min(altMin, axis.alt);
    altMax = Math.max(altMax, axis.alt);
  }

  // Trace up the left limb and back down the right to close the outline.
  return {
    outline: [...left, ...right.reverse()],
    bounds: { azMin, azMax, altMin, altMax },
  };
}

/**
 * Initial bearing (degrees true) and ground distance (metres) between two geodetic
 * positions, via the repo's standard ECEF -> ENU path — the same convention the
 * silhouette builder uses, so solver targets and these bearings cannot drift apart.
 */
export function bearingDistanceTo(from: LatLon, to: LatLon): {
  bearingDeg: number;
  distanceM: number;
} {
  // Eye heights differ by metres; the bearing impact is <0.001 deg, so compare
  // surface-level positions (h=0 on both ends).
  const a = geodeticToECEF(from.lat, from.lon, 0);
  const b = geodeticToECEF(to.lat, to.lon, 0);
  const d: [number, number, number] = [b[0] - a[0], b[1] - a[1], b[2] - a[2]];
  const enu = ecefToENU(d, from.lat, from.lon);
  const bearingDeg = (Math.atan2(enu[0], enu[1]) * 180) / Math.PI;
  return {
    bearingDeg: (bearingDeg + 360) % 360,
    distanceM: Math.hypot(enu[0], enu[1]),
  };
}

/** Build a standing point, computing bearing/distance to the landmark from coordinates. */
export function standingPoint(
  base: Omit<
    StandingPoint,
    'bearingToLandmarkDeg' | 'distanceToLandmarkM'
  >,
  landmark: LatLon,
): StandingPoint {
  const { bearingDeg, distanceM } = bearingDistanceTo(base, landmark);
  return Object.freeze({
    ...base,
    bearingToLandmarkDeg: bearingDeg,
    distanceToLandmarkM: distanceM,
  });
}
